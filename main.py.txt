def fetch_weather():
    """Fetches genuine, real-time weather for Northbrook, IL with no cloud blocks."""
    try:
        url = "https://open-meteo.com"
        res = requests.get(url, timeout=10).json()
        temp = res['current_weather']['temperature']
        return f"{temp}°F"
    except Exception:
        return "Weather currently updating"

def fetch_crypto_and_gold():
    """Pulls true live crypto prices and active spot gold metrics natively."""
    data = {"btc": 0.0, "btc_change": 0.0, "eth": 0.0, "eth_change": 0.0, "bnb": 0.0, "bnb_change": 0.0, "ratio": 0.0, "gold": "Unavailable"}
    
    # 1. Fetch genuine live prices and 24h changes from CryptoCompare
    try:
        url = "https://cryptocompare.com"
        res = requests.get(url, timeout=10).json()
        raw = res['RAW']
        
        data["btc"] = float(raw['BTC']['USD']['PRICE'])
        data["btc_change"] = float(raw['BTC']['USD']['CHANGEPCT24HOUR'])
        data["eth"] = float(raw['ETH']['USD']['PRICE'])
        data["eth_change"] = float(raw['ETH']['USD']['CHANGEPCT24HOUR'])
        data["bnb"] = float(raw['BNB']['USD']['PRICE'])
        data["bnb_change"] = float(raw['BNB']['USD']['CHANGEPCT24HOUR'])
        
        if data["btc"] > 0:
            data["ratio"] = data["bnb"] / data["btc"]
    except Exception:
        pass

    # 2. Fetch genuine live spot gold per ounce
    try:
        gold_url = "https://er-api.com"
        gold_res = requests.get(gold_url, timeout=10).json()
        gold_ounces = float(gold_res['rates']['XAU'])
        if gold_ounces > 0:
            data["gold"] = f"{1 / gold_ounces:,.2f}"
    except Exception:
        pass
        
    return data
