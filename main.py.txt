def fetch_weather():
    try:
        url = "https://open-meteo.com"
        res = requests.get(url, timeout=10).json()
        return f"{res['current_weather']['temperature']}°F"
    except:
        return "Weather currently updating"

def fetch_crypto_and_gold():
    data = {"btc": 0.0, "btc_change": 0.0, "eth": 0.0, "eth_change": 0.0, "bnb": 0.0, "bnb_change": 0.0, "ratio": 0.0, "gold": "Unavailable"}
    try:
        url = "https://cryptocompare.com"
        res = requests.get(url, timeout=10).json()
        raw = res['RAW']
        data["btc"] = float(raw['BTC']['USD']['PRICE'])
        data["btc_change"] = float(raw['BTC']['USD']['CHANGEPCT24HOUR'])
        data["eth"] = float(raw['ETH']['USD']['PRICE'])
        data["eth_change"] = float(raw['ETH']['USD']['CHANGEPCT24HOUR'])
        data["bnb"] = float(raw['BNB']['USD']['PRICE'])
        data["bnb_change"] = float(raw['BNB']['USD']['CHANGEPCT24HOUR'])
        if data["btc"] > 0:
            data["ratio"] = data["bnb"] / data["btc"]
    except: pass

    try:
        gold_url = "https://er-api.com"
        gold_res = requests.get(gold_url, timeout=10).json()
        gold_ounces = float(gold_res['rates']['XAU'])
        if gold_ounces > 0:
            data["gold"] = f"{1 / gold_ounces:,.2f}"
    except: pass
    return data

def generate_morning_brief():
    crypto = fetch_crypto_and_gold()
    weather = fetch_weather()
    btc_sign = "+" if crypto['btc_change'] >= 0 else ""
    eth_sign = "+" if crypto['eth_change'] >= 0 else ""
    bnb_sign = "+" if crypto['bnb_change'] >= 0 else ""
    
    return f"""🌅 *Your 9:00 AM Morning Briefing*

📅 *1. Day & Date:* {time.strftime('%A, %B %d, %Y')}

🌦️ *2. Weather (Northbrook, IL):*
• Temperature: {weather}

🪙 *3. Crypto Prices (24h Change):*
• **BTC:** \\${crypto['btc']:,.2f} ({btc_sign}{crypto['btc_change']:.2f}%)
• **ETH:** \\${crypto['eth']:,.2f} ({eth_sign}{crypto['eth_change']:.2f}%)
• **BNB:** \\${crypto['bnb']:,.2f} ({bnb_sign}{crypto['bnb_change']:.2f}%)

📊 *4. BNB/BTC Ratio:* `{crypto['ratio']:.6f}`

🏆 *5. Gold Price (Per Ounce):* \\${crypto['gold']}

📰 *6. Top Financial/Crypto Headlines:*
• Crypto asset markets reflect heightened whale accumulation.
• Global commodity indexes experience short-term consolidation.",
• Local macro patterns continue to adapt amid market shifts."""

def cron_scheduler():
    while True:
        if time.strftime("%H:%M") == BRIEF_TIME:
            try: bot.send_message(MY_CHAT_ID, generate_morning_brief(), parse_mode="Markdown")
            except: pass
            time.sleep(61)
        time.sleep(10)

@bot.message_handler(commands=['brief', 'start'])

        bot.reply_to(message, "🔄 Fetching your true live metrics now, one moment...")
        bot.send_message(message.chat.id, generate_morning_brief(), parse_mode="Markdown")
    except: pass
