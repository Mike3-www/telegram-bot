def fetch_weather():

def send_telegram_message(target_chat_id, text):
    """Sends message directly via raw urllib post calls."""
    try:
        url = f"{bot_url}/sendMessage"
        payload = json.dumps({"chat_id": target_chat_id, "text": text, "parse_mode": "Markdown"}).encode('utf-8')
        req = urllib.request.Request(url, data=payload, headers={'Content-Type': 'application/json', 'User-Agent': 'Mozilla/5.0'})
        with urllib.request.urlopen(req, timeout=10) as response:
            return True
    except Exception as e:
        print(f"Message Send Failure: {e}")
        return False
